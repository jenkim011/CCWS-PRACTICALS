package TYCS;

import javax.jws.WebService;
import javax.jws.WebMethod;
import javax.jws.WebParam;
@WebService(serviceName = "CricketPlayers")
public class CricketPlayers {
    @WebMethod(operationName = "hello")
    public String hello(@WebParam(name = "name") String txt) {
        return "Hello " + txt + " !";
    }
    @WebMethod(operationName = "DisplayPlayers")
    public String[] DisplayPlayers(@WebParam(name = "teamname") String teamname)
    {
            if(teamname.equalsIgnoreCase("India"))
            {
                return new String[]{"Alice","Bob","Charlie","Bumrah"};
            }
            else if(teamname.equalsIgnoreCase("Australia"))
            {
                return new String[]{"Bilal","Ansh","Vidhi","Abdurrahman"};
            }
            else
            {
                 return new String[]{"Amr","ABD","ALi","Fawzaan"};
            }

    }
}
