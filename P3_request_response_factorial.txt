Practical 3: SOAP web service - Request-Response (Factorial)

==================== FactorialService.java (package ty) ====================
package ty;

import javax.jws.WebService;
import javax.jws.WebMethod;
import javax.jws.WebParam;

@WebService(serviceName = "FactorialService")
public class FactorialService {


    @WebMethod(operationName = "fact")
    public int fact(@WebParam(name = "n") int n) {
        int f = 1;
        for (int i = 1; i <= n; i++) {
            f = f * i;
        }
        return f;
    }
}
